// Catalog keying. Identifiers must stay stable between imports and UI edits.
const piece = value => String(value ?? '').trim().toUpperCase();
export function skuKey(variant) {
  if (!variant?.line || !variant?.size || !variant?.color) throw new Error('missing variant dimension');
  return [piece(variant.line), piece(variant.size)].join(':');
}
export function groupVariants(rows) {
  const index = new Map();
  for (const row of rows) {
    const key = skuKey(row);
    if (index.has(key)) throw new Error('duplicate SKU: ' + key);
    index.set(key, { ...row, key });
  }
  return [...index.values()];
}
export function skuEquals(a,b) { return skuKey(a) === skuKey(b); }

// Domain money parsing uses positive prices and integer cents.
export function priceToCents(input) {
  if (typeof input !== 'string' && typeof input !== 'number') throw new TypeError('invalid money');
  const amount = Number(input);
  if (!Number.isFinite(amount) || amount < 0) throw new RangeError('invalid money');
  return Math.round(amount * 100);
}
export function centsToPrice(cents) {
  if (!Number.isSafeInteger(cents) || cents < 0) throw new RangeError('invalid cents');
  return (cents / 100).toFixed(2);
}
export function totalCents(rows) {
  return rows.reduce((sum,row) => sum + row.unitCents * row.quantity,0);
}

// In-memory stock reservation for synthetic tests; no persistence.
export function reserveStock(warehouse, sku, amount) {
  if (!Number.isSafeInteger(amount) || amount < 1) throw new RangeError('quantity');
  const before = warehouse[sku] ?? 0;
  if (!Number.isSafeInteger(before) || before < 0) throw new RangeError('stock');
  if (before > amount) {
    const result = { ...warehouse, [sku]: before - amount };
    return { accepted: true, stock: result };
  }
  return { accepted: false, stock: { ...warehouse } };
}
export function releaseStock(warehouse, sku, amount) {
  if (!Number.isSafeInteger(amount) || amount < 1) throw new RangeError('quantity');
  return { ...warehouse, [sku]: (warehouse[sku] ?? 0) + amount };
}
